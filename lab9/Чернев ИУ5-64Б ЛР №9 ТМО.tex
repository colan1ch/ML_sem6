\begin{lstlisting}
import numpy as np
import tensorflow as tf
from tensorflow import keras
from keras import layers
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

# Загрузка данных
X, y = load_iris(return_X_y=True)

# Нормализация
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Разделение
X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.3, random_state=42)

# One-hot кодирование целевой переменной
y_train_cat = keras.utils.to_categorical(y_train, 3)
y_test_cat = keras.utils.to_categorical(y_test, 3)
\end{lstlisting}

model_shallow = keras.Sequential([
    layers.Dense(128, activation='relu', input_shape=(4,)),
    layers.Dropout(0.2),
    layers.Dense(64, activation='relu'),
    layers.Dropout(0.2),
    layers.Dense(3, activation='softmax')
])

model_shallow.compile(optimizer='adam', 
                      loss='categorical_crossentropy',
                      metrics=['accuracy'])

history_shallow = model_shallow.fit(
    X_train, y_train_cat, epochs=100, batch_size=16, 
    validation_split=0.2, verbose=0)

\subsection*{Сравнение архитектур}

\begin{lstlisting}
# Тестирование моделей
models = {
    'Shallow (2 layers)': model_shallow,
    'Medium (3 layers)': model_medium,
    'Deep (5 layers)': model_deep
}

results = {}
for name, model in models.items():
    test_loss, test_acc = model.evaluate(X_test, y_test_cat, verbose=0)
    results[name] = test_acc
    print(f"{name}: Accuracy = {test_acc:.4f}")

# Визуализация результатов
fig, ax = plt.subplots(figsize=(10, 6))
names = list(results.keys())
accuracies = list(results.values())
colors = ['#FF6B6B', '#4ECDC4', '#45B7D1']

ax.set_ylabel('Test Accuracy')
ax.set_title('Сравнение точности архитектур на тестовой выборке')
ax.set_ylim([0.9, 1.0])
for i, (name, acc) in enumerate(zip(names, accuracies)):
    ax.text(i, acc + 0.005, f'{acc:.4f}', ha='center')
plt.xticks(rotation=15, ha='right')

\section*{Выводы}

\begin{enumerate}
    \item Глубокие сети обеспечивают более точное предсказание благодаря большей емкости
    \item Dropout-слои критичны для предотвращения переобучения
    \item Для датасета Iris даже мелкая сеть показывает хорошие результаты
    \item Глубокие сети требуют больше данных для избежания переобучения
    \item Батч-нормализация может улучшить сходимость обучения
    \item t-SNE лучше сохраняет локальную структуру данных чем PCA
    \item Максимальная вероятность предсказания может использоваться как мера уверенности модели
\end{enumerate}
