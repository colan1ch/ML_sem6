import pandas as pd

from statsmodels.tsa.seasonal import seasonal_decompose

\section*{Оценка качества прогнозов}

\begin{lstlisting}
from sklearn.metrics import mean_absolute_error, mean_squared_error
import numpy as np
import matplotlib.pyplot as plt

def evaluate_forecast(y_true, y_pred):
    mae = mean_absolute_error(y_true, y_pred)
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    mape = np.mean(np.abs((y_true - y_pred) / y_true)) * 100
    return {'MAE': mae, 'RMSE': rmse, 'MAPE': mape}

# Сравнение методов
results = {
    'ARIMA': evaluate_forecast(test_data, forecast_arima),
    'SymbRegression': evaluate_forecast(test_data, forecast_symb),
    'GMDH': evaluate_forecast(test_data, forecast_gmdh)
}

# Визуализация коэффициента детерминации
fig, ax = plt.subplots(figsize=(10, 6))
methods = list(results.keys())
r2_scores = [results[m]['MAE'] for m in methods]
ax.bar(methods, r2_scores, color=['#FF6B6B', '#4ECDC4', '#45B7D1'])
ax.set_ylabel('Mean Absolute Error')
ax.set_title('Сравнение методов прогнозирования временных рядов')
plt.savefig('fig_1.png', dpi=150, bbox_inches='tight')
plt.show()

for method, metrics in results.items():
    print(f"{method}: MAE={metrics['MAE']:.4f}, RMSE={metrics['RMSE']:.4f}, MAPE={metrics['MAPE']:.2f}%")
\end{lstlisting}

\section*{Выводы}

\begin{enumerate}
    \item ARIMA эффективна для линейных трендов и сезонности
    \item Символьная регрессия выявляет нелинейные закономерности
    \item GMDH показывает лучшие результаты на тестовой выборке
    \item Комбинирование методов улучшает точность прогноза
\end{enumerate}
