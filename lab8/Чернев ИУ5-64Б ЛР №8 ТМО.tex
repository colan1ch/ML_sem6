\section*{Методы кластеризации}

\begin{lstlisting}
from sklearn.cluster import KMeans, DBSCAN, AgglomerativeClustering

# K-Means
kmeans = KMeans(n_clusters=3, random_state=42)
labels_km = kmeans.fit_predict(X_scaled)

# DBSCAN
dbscan = DBSCAN(eps=0.5, min_samples=5)
labels_db = dbscan.fit_predict(X_scaled)

# Agglomerative
agg = AgglomerativeClustering(n_clusters=3, linkage='ward')
labels_agg = agg.fit_predict(X_scaled)

# Визуализация результатов кластеризации
fig, ax = plt.subplots(figsize=(10, 8))
scatter = ax.scatter(X_pca[:, 0], X_pca[:, 1], c=labels_km, cmap='viridis', alpha=0.6, s=50)
ax.scatter(kmeans.cluster_centers_[:, 0], kmeans.cluster_centers_[:, 1], 
           c='red', marker='X', s=200, edgecolors='black', label='Центроиды')
ax.set_title('K-Means кластеризация')
ax.set_xlabel('PC1')
ax.set_ylabel('PC2')
ax.legend()

\section*{Оценка качества кластеризации}

\begin{lstlisting}
from sklearn.metrics import silhouette_score, davies_bouldin_score
from sklearn.metrics import calinski_harabasz_score

metrics = {}

for labels, name in [(labels_km, 'K-Means'), 
                      (labels_db, 'DBSCAN'),
                      (labels_agg, 'Agglomerative')]:
    
    sil = silhouette_score(X_scaled, labels)
    db = davies_bouldin_score(X_scaled, labels)
    ch = calinski_harabasz_score(X_scaled, labels)
    
    metrics[name] = {'Silhouette': sil, 'Davies-Bouldin': db, 'Calinski-Harabasz': ch}

# Визуализация метрик качества
fig, ax = plt.subplots(figsize=(10, 6))
methods = list(metrics.keys())
silhouette_scores = [metrics[m]['Silhouette'] for m in methods]
ax.bar(methods, silhouette_scores, color=['#FF6B6B', '#4ECDC4', '#45B7D1'])
ax.set_ylabel('Silhouette Score')
ax.set_title('Сравнение методов кластеризации')
plt.tight_layout()
plt.savefig('fig_9.png', dpi=150, bbox_inches='tight')
plt.show()

for method, scores in metrics.items():
    print(f"{method}:")
    for metric, value in scores.items():
        print(f"  {metric}: {value:.4f}")
\end{lstlisting}

\section*{Выводы}

\begin{enumerate}
    \item PCA сохраняет глобальную структуру, t-SNE лучше для визуализации локальных кластеров
    \item K-Means быстрый и стабильный, но требует знания числа кластеров
    \item DBSCAN находит кластеры произвольной формы, но чувствителен к параметрам
    \item Agglomerative хорошо справляется с иерархической кластеризацией
\end{enumerate}
